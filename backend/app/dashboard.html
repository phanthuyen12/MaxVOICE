Hướng dẫn tích hợp API (Port 8686)

Sử dụng header X-API-Key: kv_live_... khi gửi request từ hệ thống bên ngoài

# 1. Truyền Text -> Nhận trực tiếp file âm thanh Giọng nói (.mp3)
curl -X POST "http://localhost:8686/api/v1/tts" \
  -H "X-API-Key: YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "text": "Xin chào, đây là giọng đọc tạo từ API KingVoice",
    "language": "vi",
    "voice_id": "multi:vi-VN-NamMinh",
    "output_format": "mp3"
  }' --output voice.mp3

# 2. Nhận diện giọng nói từ Audio/Video -> Tự dịch & Tạo ra Giọng AI mới
curl -X POST "http://localhost:8686/api/v1/audio/dub" \
  -H "X-API-Key: YOUR_API_KEY" \
  -F "file=@input_video_or_audio.mp4" \
  -F "target_language=vi" \
  -F "voice_id=multi:vi-VN-HoaiMy"

# 3. Lồng tiếng Video hoàn chỉnh (Cắt + Whisper STT + Dịch + TTS + Ghép MP4)
curl -X POST "http://localhost:8686/api/v1/videos/generate" \
  -H "X-API-Key: YOUR_API_KEY" \
  -F "files=@input.mp4" \
  -F "target_language=ru" \
  -F "voice_id=multi:ru-RU-Dmitry" \
  -F "burn_subtitles=true"
